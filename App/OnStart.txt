// ============================================================
// App.OnStart
// In Studio: select the "App" object at the top of the Tree view,
// pick the OnStart property in the dropdown, and paste everything
// in the block below (comments are allowed in Power Fx).
//
// This preloads the collections so the first screen renders with
// data immediately. The screens still refresh these collections in
// their own OnVisible, so data stays current as you navigate.
// ============================================================

// Current user info (handy for formulas and debugging)
Set(varUserName, User().FullName);
Set(varUserEmail, Lower(User().Email));

// ------------------------------------------------------------
// App link + deep linking (Share Event feature).
// varAppURL: after the FIRST publish, get the app's web link
// (make.powerapps.com > Apps > ... on the app > Details > Web link)
// and paste it between the quotes below, then publish again.
// The Share Event modal appends ?eventid=<ID> to this link.
// When someone opens such a link, the "eventid" parameter below
// routes them straight to that event's detail page (also set the
// App's StartScreen property - see the note at the very bottom).
// ------------------------------------------------------------
Set(varAppURL, "");
If(!IsBlank(Param("eventid")),
    Set(varDetailEventID, Value(Param("eventid")));
    Set(varDeepEvent, true));

// ------------------------------------------------------------
// Role lookup (EventSessionRegistration_AppRoles list).
//   UserEmail = Person column (people picker - no typos possible)
//   Role      = Choice column with exactly two choices: Admin, Coordinator
// The person-column comparison isn't delegable, so the list is collected
// first and the lookup runs in memory - no delegation warning.
// Bootstrap: while EventSessionRegistration_AppRoles has no rows yet, everyone is treated
// as an Admin so you can't lock yourself out. Add your own row
// (Role = Admin) first, then add the others.
// ------------------------------------------------------------
// Collect the tiny roles list first, then look up in memory - collections
// aren't delegable sources, so this raises no delegation warning (the list
// stays far below the data-row limit).
ClearCollect(colAppRoles, EventSessionRegistration_AppRoles);
Set(varMyRole, LookUp(colAppRoles, Lower(UserEmail.Email) = Lower(User().Email), Role.Value));
Set(varRolesConfigured, !IsEmpty(colAppRoles));
Set(varIsAdmin, If(varRolesConfigured, varMyRole = "Admin", true));
Set(varIsCoordinator, varIsAdmin || varMyRole = "Coordinator");

// Sidebar default: open for admins/coordinators, collapsed for everyone else.
// Every screen has a ☰ button in its header to toggle it at any time.
Set(varSidebarOpen, varIsCoordinator);

// This week's active time slots (delegable filter)
ClearCollect(
    colWeekSlots,
    Filter(
        EventSessionRegistration_SessionTimeSlots,
        IsActive = true,
        SessionDate >= Today(),
        SessionDate < DateAdd(Today(), 7, TimeUnit.Days)
    )
);

// All future confirmed registrations (delegable filter)
ClearCollect(
    colFutureRegs,
    Filter(
        EventSessionRegistration_Registrations,
        Status = "Confirmed",
        SessionDate >= Today()
    )
);

// Current user's confirmed registrations (delegable filter)
ClearCollect(
    colMyRegsAll,
    Filter(
        EventSessionRegistration_Registrations,
        UserEmail = Lower(User().Email),
        Status = "Confirmed"
    )
)
