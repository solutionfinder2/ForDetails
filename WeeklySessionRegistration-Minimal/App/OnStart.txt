// ============================================================
// App.OnStart - MINIMAL EDITION (3 screens, no EmailTemplates)
// In Studio: select the "App" object at the top of the Tree view,
// pick the OnStart property in the dropdown, and paste everything
// in the block below (comments are allowed in Power Fx).
//
// Screens in this edition: scrIntro, scrEventQuickReg, scrAdmin.
// SharePoint lists used:   EventSessionRegistration_Events,
//   EventSessionRegistration_SessionTimeSlots,
//   EventSessionRegistration_Registrations,
//   EventSessionRegistration_AppRoles.
// The EmailTemplates list is NOT used - emails are hardcoded in
// the screens; import the NoTemplates flow package.
// ============================================================

// Current user info (handy for formulas and debugging)
Set(varUserName, User().FullName);
Set(varUserEmail, Lower(User().Email));

// ------------------------------------------------------------
// PROMOTED (FEATURED) EVENT - shown on the Intro page.
// There is no Settings screen in this edition; pick ONE of these:
//
//   Option A (no republish needed - recommended):
//     Leave the value below at 0 and set the "IsPromoted" column
//     to Yes on the event item in the EventSessionRegistration_Events
//     list (SharePoint > list > edit the item). Remember to turn it
//     off on the old event when you promote a new one.
//
//   Option B (pin by ID):
//     Put the event's ID (the SharePoint item ID from the Events
//     list) in the line below and republish the app. While this is
//     greater than 0 it wins over the IsPromoted column.
// ------------------------------------------------------------
Set(varPromotedEventID, 0);

// ------------------------------------------------------------
// App link + deep linking (Share Event feature).
// varAppURL: after the FIRST publish, get the app's web link
// (make.powerapps.com > Apps > ... on the app > Details > Web link)
// and paste it between the quotes below, then publish again.
// The Share Event modal appends ?eventid=<ID> to this link.
// When someone opens such a link, the "eventid" parameter below
// routes them straight to the event page (also set the App's
// StartScreen property - see the note at the very bottom).
// ------------------------------------------------------------
Set(varAppURL, "");
If(!IsBlank(Param("eventid")),
    Set(varDetailEventID, Value(Param("eventid")));
    Set(varDeepEvent, true));

// ------------------------------------------------------------

// ------------------------------------------------------------
// Registrations list metadata (EventSessionRegistration_Registrations).
// Beyond Title / SlotID / SessionDate / TimeSlot / EventID / EventName /
// UserName / UserEmail / Status, the app also collects:
//   JobTitle              = Single line of text
//   ExperienceLevel       = Choice: Beginner, Intermediate, Advanced
//   AttendancePreference  = Choice: In-Person, Virtual, No Preference
//   OrgUnit               = Choice: HR, IT, Finance, Operations, Other
//   HeardFrom             = Choice: Manager, Email, Teams, Colleague, Other
// Create these columns in SharePoint first (see SharePoint/REGISTRATION-METADATA.md),
// then refresh the data source in Studio.
// ------------------------------------------------------------

// Role lookup (EventSessionRegistration_AppRoles list).
//   UserEmail = Person column (people picker - no typos possible)
//   Role      = Choice column with exactly two choices: Admin, Coordinator
// The person-column comparison isn't delegable, so the tiny list is
// collected first and the lookup runs in memory - no delegation warning.
// Bootstrap: while the list has no rows yet, everyone is treated as an
// Admin so you can't lock yourself out. Add your own row (Role = Admin)
// first, then add the others.
// ------------------------------------------------------------
ClearCollect(colAppRoles, EventSessionRegistration_AppRoles);
Set(varMyRole, LookUp(colAppRoles, Lower(UserEmail.Email) = Lower(User().Email), Role.Value));
Set(varRolesConfigured, !IsEmpty(colAppRoles));
Set(varIsAdmin, If(varRolesConfigured, varMyRole = "Admin", true));
Set(varIsCoordinator, varIsAdmin || varMyRole = "Coordinator");

// Sidebar default on the Admin screen: open for admins/coordinators.
Set(varSidebarOpen, varIsCoordinator);

// Preload future confirmed registrations (used by the event page's
// seat counts; the screens also refresh this in their own OnVisible).
ClearCollect(
    colFutureRegs,
    Filter(
        EventSessionRegistration_Registrations,
        Status = "Confirmed",
        SessionDate >= Today()
    )
)
